# Project 2a, Plan C: CLI (`dndm server`, `dndm connect`) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a person or an agent run the server and play through it from the terminal: `dndm server migrate | create-channel | start`, and `dndm connect` as the streamer or as any number of dev viewers, in script and interactive modes. CI gains Postgres.

**Architecture:** `apps/cli` depends on `@dndm/server` for the server subcommands. `dndm connect` uses a small client (`apps/cli/src/client.ts`) over `fetch` and Node's built-in `WebSocket` (Node ≥ 24), so no extra dependency. Scripts keep `dndm run`'s `{as, command}` lines. Real-time pauses are `{pause}`, and `{wait}` and `{next}` are refused with a hint (spec §8.2).

**Tech Stack:** As before. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-10-11-server-core-design.md` (rev 7), §1.1, §8. Flagged decisions: `docs/superpowers/plans/2026-10-11-project-2a-README.md`.

**Prerequisite:** Plans A and B complete.

## Global Constraints

- Settings come from `DATABASE_URL`, `DNDM_SECRET` and `DNDM_DEV` (spec §8.1). A missing setting is a clean CLI error (exit 2), never a stack trace.
- `create-channel` without `--seed` uses a random seed (§8.1).
- `dndm connect` rejects `{wait}` and `{next}` with a message that says to use `pause` (§8.2).
- `--json` prints one JSON object per line, as every other `dndm` command does.
- Every CLI error prints `error: <message>` (or `{"kind":"error","text":…}` with `--json`) and exits non-zero.

## Review Focus

1. **`dndm connect` pointed at a server that isn't running**: a person expects "can't reach the server at <url>", not a fetch stack trace. Test in Task 2.
2. **A connect script line with `as` set to a viewer while the server isn't in dev mode**: a person expects a clear "dev login is off on this server" error. Test in Task 2.
3. **`create-channel` with a config that has errors**: a person expects the validation issues listed and exit 1, with no channel created. Test in Task 1.
4. **`{"pause":"events"}` when no event ever comes**: a person expects it to give up after a timeout with a note, not hang forever. Test in Task 2.
5. **A missing `DNDM_SECRET`**: a person expects `error: set DNDM_SECRET (16+ characters)` and exit 2. Test in Task 1.

```ts
import { randomInt } from 'node:crypto';
import { resolve } from 'node:path';
import { formatIssue, ValidationError } from '@dndm/engine';
import { loadChannel } from '@dndm/pack-loader';
import { createChannel, defaultConfig, migrate, officialBundles, Secrets, startServer, Store } from '@dndm/server';
import { CliError } from './ids';

export interface ServerCliOpts {
  env: NodeJS.ProcessEnv; json?: boolean; port?: string; host?: string; config?: string; seed?: number;
  write: (line: string) => void;
  /** Resolves when `start` should stop. Defaults to SIGINT or SIGTERM. */
  until?: Promise<void>;
}

function need(env: NodeJS.ProcessEnv, name: 'DATABASE_URL' | 'DNDM_SECRET'): string {
  const v = env[name];
  if (!v) throw new CliError(name === 'DNDM_SECRET' ? 'set DNDM_SECRET (16+ characters)' : 'set DATABASE_URL (for example postgres://dndm:dndm@localhost:5455/dndm)');
  if (name === 'DNDM_SECRET' && v.length < 16) throw new CliError('DNDM_SECRET must be at least 16 characters');
  return v;
}

const stopSignal = () => new Promise<void>((done) => {
  process.once('SIGINT', () => done());
  process.once('SIGTERM', () => done());
});

/** `dndm server <migrate|create-channel|start>` (spec 2a §8.1). */
export async function serverCli(sub: string | undefined, args: string[], o: ServerCliOpts): Promise<number> {
  const say = (kind: string, text: string, extra: Record<string, unknown> = {}) =>
    o.write(o.json ? JSON.stringify({ kind, text, ...extra }) : text);

  switch (sub) {
    case 'migrate': {
      await migrate(need(o.env, 'DATABASE_URL'));
      say('note', 'Migrations applied.');
      return 0;
    }
    case 'create-channel': {
      const name = args[0];
      if (!name) throw new CliError('Usage: dndm server create-channel <name> [--config file] [--seed n]');
      const url = need(o.env, 'DATABASE_URL');
      const secrets = new Secrets(need(o.env, 'DNDM_SECRET'));
      const config = (o.config ? loadChannel(resolve(o.config)) : defaultConfig()) as Record<string, unknown>;
      const store = Store.open(url);
      try {
        const seed = o.seed ?? randomInt(0, 2 ** 31);
        const made = await createChannel({ store, secrets, bundles: await officialBundles(), name, config, seed });
        if (o.json) {
          o.write(JSON.stringify({ kind: 'channel', name, ...made, seed }));
        } else {
          o.write(`Channel ${name} created (seed ${seed}).`);
          o.write(`  channel: ${made.id}`);
          o.write(`  dock token (keep it secret, it runs the game): ${made.dockToken}`);
          o.write(`  overlay token: ${made.overlayToken}`);
          o.write(`Connect as the streamer: dndm connect http://127.0.0.1:8787 --channel ${made.id} --dock ${made.dockToken}`);
        }
        return 0;
      } catch (e) {
        if (e instanceof ValidationError) { e.issues.forEach((i) => o.write(formatIssue(i))); return 1; }
        throw e;
      } finally {
        await store.close();
      }
    }
    case 'start': {
      const port = o.port !== undefined ? Number(o.port) : 8787;
      if (!Number.isInteger(port) || port < 0 || port > 65535) throw new CliError('--port must be a whole number from 0 to 65535');
      const dev = o.env.DNDM_DEV === '1';
      const srv = await startServer({
        databaseUrl: need(o.env, 'DATABASE_URL'), secret: need(o.env, 'DNDM_SECRET'), dev, port, host: o.host,
        log: (msg, extra) => o.write(JSON.stringify({ kind: 'log', text: msg, ...extra })),
      });
      say('note', `dndm server listening on ${srv.url}${dev ? ' (dev login on)' : ''}`, { url: srv.url });
      await (o.until ?? stopSignal());
      await srv.close();
      return 0;
    }
    default:
      throw new CliError('Usage: dndm server <migrate | create-channel <name> | start [--port n]>');
  }
}
```

```ts
import { randomUUID } from 'node:crypto';
import type { Description, View } from '@dndm/engine';
import type { Answer, ServerMessage } from '@dndm/protocol';
import { CliError } from './ids';

export interface Live {
  welcome: ServerMessage;
  on(fn: (m: ServerMessage) => void): () => void;
  /** Resolves true when the next `events` message arrives, false after timeoutMs. */
  nextEvents(timeoutMs: number): Promise<boolean>;
  close(): void;
}

/** A small HTTP + WebSocket client for one channel (spec 2a §4, §5). */
export class Client {
  constructor(readonly base: string, readonly channel: string) {}

  private async call(path: string, init: { method: string; token?: string; body?: unknown }): Promise<{ status: number; body: any }> {
    let res: Response;
    try {
      res = await fetch(new URL(path, this.base), {
        method: init.method,
        headers: {
          ...(init.body !== undefined ? { 'content-type': 'application/json' } : {}),
          ...(init.token ? { authorization: `Bearer ${init.token}` } : {}),
        },
        body: init.body !== undefined ? JSON.stringify(init.body) : undefined,
      });
    } catch {
      throw new CliError(`can't reach the server at ${this.base}`);
    }
    return { status: res.status, body: await res.json().catch(() => null) };
  }

  async login(viewer: string): Promise<string> {
    const r = await this.call('/dev/login', { method: 'POST', body: { channel: this.channel, viewer } });
    // Without dev mode the route doesn't exist, so Fastify answers its own 404 (no `ok` field).
    if (r.status === 404 && r.body?.ok === undefined) throw new CliError('dev login is off on this server (start it with DNDM_DEV=1)');
    if (!r.body?.ok) throw new CliError(`dev login for ${viewer} failed: ${r.body?.error?.message ?? r.status}`);
    return r.body.token as string;
  }

  async command(token: string, command: Record<string, unknown>, id: string = randomUUID()): Promise<Answer> {
    const r = await this.call(`/channels/${this.channel}/commands`, { method: 'POST', token, body: { id, command } });
    if (!r.body) throw new CliError(`the server answered ${r.status} with no body`);
    return r.body as Answer;
  }

  async view(token: string): Promise<View> {
    const r = await this.call(`/channels/${this.channel}/view`, { method: 'GET', token });
    if (!r.body?.ok) throw new CliError(r.body?.error?.message ?? `view failed (${r.status})`);
    return r.body.view as View;
  }

  async describe(token: string): Promise<Description> {
    const r = await this.call(`/channels/${this.channel}/describe`, { method: 'GET', token });
    if (!r.body?.ok) throw new CliError(r.body?.error?.message ?? `describe failed (${r.status})`);
    return r.body.describe as Description;
  }

  live(token: string): Promise<Live> {
    const url = new URL(`/channels/${this.channel}/live`, this.base);
    url.protocol = url.protocol === 'https:' ? 'wss:' : 'ws:';
    return new Promise((resolve, reject) => {
      let ws: WebSocket;
      try { ws = new WebSocket(url); } catch { reject(new CliError(`can't reach the server at ${this.base}`)); return; }
      const listeners = new Set<(m: ServerMessage) => void>();
      let welcomed = false;
      ws.addEventListener('open', () => ws.send(JSON.stringify({ type: 'hello', token })));
      ws.addEventListener('error', () => { if (!welcomed) reject(new CliError(`can't reach the server at ${this.base}`)); });
      ws.addEventListener('close', (ev) => { if (!welcomed) reject(new CliError(`the server refused the live connection (${ev.code} ${ev.reason})`)); });
      ws.addEventListener('message', (ev) => {
        const m = JSON.parse(String(ev.data)) as ServerMessage;
        if (!welcomed && m.type === 'welcome') {
          welcomed = true;
          resolve({
            welcome: m,
            on: (fn) => { listeners.add(fn); return () => { listeners.delete(fn); }; },
            nextEvents: (timeoutMs) => new Promise((done) => {
              const fn = (x: ServerMessage) => {
                if (x.type !== 'events') return;
                clearTimeout(timer);
                listeners.delete(fn);
                done(true);
              };
              const timer = setTimeout(() => { listeners.delete(fn); done(false); }, timeoutMs);
              listeners.add(fn);
            }),
            close: () => ws.close(),
          });
          return;
        }
        for (const fn of listeners) fn(m);
      });
    });
  }
}
```

```ts
import { readFileSync } from 'node:fs';
import type { Readable, Writable } from 'node:stream';

import type { Answer, ServerMessage } from '@dndm/protocol';
import { Client, type Live } from './client';

import { CliError } from './ids';

export type ConnectLine = { pause: string | number } | { as: string; command: Record<string, unknown> & { type: string } };

const isObj = (v: unknown): v is Record<string, unknown> => typeof v === 'object' && v !== null && !Array.isArray(v);
export const STEPPED = '`wait` and `next` move the stepped clock in `dndm run`; a live server\'s clock can\'t be moved. Use {"pause":"5s"} or {"pause":"events"}';

/** Spec 2a §8.2: `dndm run`'s lines, with real-time pauses instead of stepped waits. */
export function parseConnectLine(text: string, lineNo: number): ConnectLine {
  let v: unknown;
  try { v = JSON.parse(text); } catch (e) { throw new CliError(`line ${lineNo}: not JSON (${(e as Error).message})`); }
  if (isObj(v) && ('wait' in v || 'next' in v)) throw new CliError(`line ${lineNo}: ${STEPPED}`);
  if (isObj(v) && 'pause' in v && (typeof v.pause === 'string' || typeof v.pause === 'number')) {
    if (v.pause !== 'events') {
      try { parseDuration(v.pause); } catch (e) { throw new CliError(`line ${lineNo}: ${(e as Error).message}`); }
    }
    return { pause: v.pause };
  }
  if (isObj(v) && typeof v.as === 'string' && isObj(v.command) && typeof v.command.type === 'string') {
    return { as: v.as, command: v.command as Record<string, unknown> & { type: string } };
  }
  throw new CliError(`line ${lineNo}: expected {"pause"} or {"as", "command"}`);
}

export interface ConnectOpts {
  url: string; channel?: string; dock?: string; viewer?: string; script?: string; json?: boolean;
  timeoutMs?: number; input: Readable; output: Writable;
}

/** One connection's identities: the dock token for "streamer", and dev logins for viewer names. */
export class Session {
  private tokens = new Map<string, string>();
  live: Live | null = null;

  constructor(readonly client: Client, readonly o: ConnectOpts, private readonly write: (o: Output) => void) {
    if (o.dock) this.tokens.set('streamer', o.dock);
  }

  async token(as: string): Promise<string> {
    const known = this.tokens.get(as);
    if (known) return known;
    if (as === 'streamer') throw new CliError('streamer lines need --dock <token>');
    const t = await this.client.login(as);
    this.tokens.set(as, t);
    return t;
  }

  /** Watches as the streamer when there's a dock token (it sees every event), else as --viewer. */
  async open(): Promise<void> {
    const who = this.o.dock ? 'streamer' : this.o.viewer;
    if (!who) throw new CliError('give --dock <token> or --viewer <name>');
    this.live = await this.client.live(await this.token(who));
    this.live.on((m: ServerMessage) => {
      if (m.type === 'events') for (const e of m.events) this.write(eventOut(e));
      if (m.type === 'config-changed') this.write({ kind: 'note', at: 0, text: `Config changed (version ${m.configVersion})` });
    });
  }

  async send(as: string, command: Record<string, unknown> & { type: string }): Promise<Answer> {
    const { character: _ignored, ...rest } = command;
    const a = await this.client.command(await this.token(as), as === 'streamer' ? command : rest);
    if (!a.ok) this.write({ kind: 'rejected', at: 0, as, code: a.error.code, text: a.error.message });
    return a;
  }

  async pause(p: string | number): Promise<void> {
    if (p === 'events') {
      const got = await this.live!.nextEvents(this.o.timeoutMs ?? 30_000);
      if (!got) this.write({ kind: 'note', at: 0, text: 'No events arrived before the timeout' });
      return;
    }
    await new Promise((r) => setTimeout(r, parseDuration(p)));
  }

  close(): void {
    this.live?.close();
  }
}

const eventOut = (e: GameEvent): Output => ({ kind: 'event', at: e.at, text: e.text, event: e });

export async function connect(o: ConnectOpts): Promise<number> {
  if (!o.channel) throw new CliError('give --channel <id>');
  const writeLine = (line: string) => o.output.write(`${line}\n`);
  const write = (x: Output) => writeLine(formatOutput(x, !!o.json));
  const s = new Session(new Client(o.url, o.channel), o, write);
  try {
    if (!o.script) {
      const { interactive } = await import('./connect-interactive');
      await s.open();
      return await interactive(s, o, write, writeLine);
    }
    let text: string;
    try { text = readFileSync(o.script, 'utf8'); } catch { throw new CliError(`can't read ${o.script}`); }
    const lines = text.split('\n').map((l, i) => [l, i + 1] as const).filter(([l]) => l.trim());
    const parsed = lines.map(([l, n]) => parseConnectLine(l, n));
    await s.open();
    for (const line of parsed) {
      if ('pause' in line) await s.pause(line.pause);
      else await s.send(line.as, line.command);
    }
    await new Promise((r) => setTimeout(r, 100));
    return 0;
  } catch (e) {
    if (e instanceof CliError) {
      writeLine(o.json ? JSON.stringify({ kind: 'error', text: e.message }) : `error: ${e.message}`);
      return 1;
    }
    throw e;
  } finally {
    s.close();
  }
}
```

`apps/cli/src/connect-interactive.ts`:

```ts
import { createInterface } from 'node:readline';
import { STEPPED, type ConnectOpts, type Session } from './connect';
import type { Output } from './driver';
import { CliError } from './ids';
import { HELP, parseLine } from './parse';
import { renderState, renderView } from './render';

export async function interactive(s: Session, o: ConnectOpts, write: (o: Output) => void, writeLine: (line: string) => void): Promise<number> {
  const json = !!o.json;
  const desc = await s.client.describe(await s.token(o.dock ? 'streamer' : o.viewer!));
  const tty = !json && 'isTTY' in o.input && !!(o.input as { isTTY?: boolean }).isTTY;
  const note = (text: string) => writeLine(json ? JSON.stringify({ kind: 'error', text }) : `! ${text}`);
  if (!json) writeLine('dndm connect: live. Type help for commands. Time runs on its own; use pause <duration|events> to wait.');
  const rl = createInterface({ input: o.input, ...(tty ? { output: o.output, terminal: true } : {}) });
  if (tty) { rl.setPrompt('> '); rl.prompt(); }

  for await (const text of rl) {
    let quit = false;
    try {
      const trimmed = text.trim();
      if (trimmed.startsWith('pause')) {
        const arg = trimmed.split(/\s+/)[1];
        if (!arg) throw new CliError('Usage: pause <duration|events>');
        await s.pause(arg);
      } else {
        const action = parseLine(text, desc);
        if (action) {
          switch (action.kind) {
            case 'quit': quit = true; break;
            case 'help': if (!json) writeLine(`${HELP}\n  pause <duration|events>    wait in real time (wait and next don't work live)`); break;
            case 'describe': writeLine(JSON.stringify(desc, null, json ? undefined : 2)); break;
            case 'state': {
              const v = await s.client.view(await s.token(o.dock ? 'streamer' : o.viewer!));
              if (json) writeLine(JSON.stringify({ kind: 'state', ...v })); else renderState(v).forEach(writeLine);
              break;
            }
            case 'view': {
              const v = await s.client.view(await s.token(action.id));
              if (json) writeLine(JSON.stringify({ kind: 'view', ...v })); else renderView(v).forEach(writeLine);
              break;
            }
            case 'script': {
              const line = action.line;
              if ('wait' in line || 'next' in line) throw new CliError(STEPPED);
              await s.send(line.as, line.command as Record<string, unknown> & { type: string });
              await new Promise((r) => setTimeout(r, 50));
              break;
            }
          }
        }
      }
    } catch (e) {
      if (e instanceof CliError) note(e.message);
      else throw e;
    }
    if (quit) break;
    if (tty) rl.prompt();
  }
  rl.close();
  return 0;
}
```
